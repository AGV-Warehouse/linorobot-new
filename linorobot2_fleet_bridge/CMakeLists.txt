cmake_minimum_required(VERSION 3.16)
project(linorobot2_fleet_bridge)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_INSTALL_RPATH_USE_LINK_PATH TRUE)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(nlohmann_json REQUIRED)
find_package(SQLite3 REQUIRED)
add_library(fleet_logic src/execution_journal.cpp src/route_guard.cpp src/robot_state.cpp src/vda_actions.cpp)
target_include_directories(fleet_logic PUBLIC include)
target_link_libraries(fleet_logic PUBLIC SQLite::SQLite3 nlohmann_json::nlohmann_json)
option(FLEET_CORE_ONLY "Build platform-neutral execution tests without ROS" OFF)
if(FLEET_CORE_ONLY)
  enable_testing()
  find_package(GTest REQUIRED)
  foreach(name order_execution recovery state_mapping runtime_policy)
    add_executable(test_${name} test/test_${name}.cpp)
    target_link_libraries(test_${name} fleet_logic GTest::gtest_main)
    add_test(NAME ${name} COMMAND test_${name})
  endforeach()
  return()
endif()
find_package(ament_cmake REQUIRED)
find_package(vda5050_core REQUIRED)
foreach(dependency rclcpp rclcpp_action nav2_msgs nav_msgs geometry_msgs sensor_msgs std_msgs std_srvs tf2_ros)
  find_package(${dependency} REQUIRED)
endforeach()
add_executable(fleet_bridge src/bridge_node.cpp src/nav2_executor.cpp src/mqtt_transport.cpp)
target_link_libraries(fleet_bridge fleet_logic vda5050_core::execution)
ament_target_dependencies(fleet_bridge rclcpp rclcpp_action nav2_msgs nav_msgs geometry_msgs sensor_msgs std_msgs std_srvs tf2_ros)
install(TARGETS fleet_bridge DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  foreach(name order_execution recovery state_mapping runtime_policy)
    ament_add_gtest(test_${name} test/test_${name}.cpp)
    target_link_libraries(test_${name} fleet_logic)
  endforeach()
endif()
ament_package()
